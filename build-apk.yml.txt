# GitHub Actions: сборка APK + публикация релиза (для автообновления).
# Файл назван .txt намеренно: токен песочницы не может пушить в .github/workflows/.
# ОДИН РАЗ через веб-интерфейс GitHub: Add file → Create new file →
# имя: .github/workflows/build.yml → вставить всё содержимое ниже → Commit.
# После этого каждый push в main собирает свежий APK и обновляет релиз:
# Actions → последний зелёный запуск → Artifacts → app-debug-apk.
# ============================================================
name: Build APK and Release
on:
  push:
    branches: [main]
  workflow_dispatch:

permissions:
  contents: write

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 20
      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: 17
      - run: npm install
      # Номер сборки заранее — он нужен и приложению, и релизу
      - id: ver
        run: echo "tag=v$(node -p "require('./package.json').version")-b${{ github.run_number }}" >> "$GITHUB_OUTPUT"
      # Вшиваем полную версию со сборкой в приложение, иначе автообновление будет
      # бесконечно предлагать "новую" версию (приложение не знало свой build-номер)
      - run: sed -i "s|^export const APP_VERSION = '.*';|export const APP_VERSION = '${{ steps.ver.outputs.tag }}';|" www/js/config.js
      - run: grep "APP_VERSION" www/js/config.js
      - run: npx cap add android      # создаёт android/ (в .gitignore)
      # Подкладываем нативный модуль чтения уведомлений и свою MainActivity
      - run: |
          mkdir -p android/app/src/main/java/com/finanalyzer/app
          cp native/NotificationListenerPlugin.java native/BankNotificationService.java native/MainActivity.java android/app/src/main/java/com/finanalyzer/app/
      # Регистрируем службу чтения уведомлений в манифесте
      - run: |
          python3 - <<'EOF'
          p = 'android/app/src/main/AndroidManifest.xml'
          s = open(p, encoding='utf-8').read()
          svc = ('<service android:name=".BankNotificationService" '
                 'android:label="FinAnalyzer" '
                 'android:permission="android.permission.BIND_NOTIFICATION_LISTENER_SERVICE" '
                 'android:exported="true">'
                 '<intent-filter><action android:name="android.service.notification.NotificationListenerService" /></intent-filter>'
                 '</service>')
          assert '</application>' in s
          s = s.replace('</application>', svc + '</application>')
          open(p, 'w', encoding='utf-8').write(s)
          EOF
      - run: npx cap sync android      # копирует www/ в android-проект
      - run: cd android && chmod +x gradlew && ./gradlew assembleDebug --stacktrace
      - uses: actions/upload-artifact@v4
        with:
          name: app-debug-apk
          path: android/app/build/outputs/apk/debug/app-debug.apk
      # Релиз для автообновления: приложение на телефоне смотрит releases/latest
      - uses: softprops/action-gh-release@v2
        with:
          tag_name: ${{ steps.ver.outputs.tag }}
          name: Финансовый анализатор ${{ steps.ver.outputs.tag }}
          body: |
            Автосборка из коммита ${{ github.sha }}.
            Установка: скачать APK → открыть → разрешить установку из источника.
            Приложение само предложит обновление при выходе новой версии.
          files: android/app/build/outputs/apk/debug/app-debug.apk
          draft: false
          prerelease: false
        env:
          GITHUB_TOKEN: ${{ secrets.GITHUB_TOKEN }}
