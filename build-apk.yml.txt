# GitHub Actions: сборка APK + публикация релиза (для автообновления).
# ⚠️ GitHub не даёт ботам пушить файлы прямо в .github/workflows/ (нет workflow-scopes).
# Поэтому файл лежит здесь. ОДИН РАЗ: скопируй его содержимое в
# .github/workflows/build.yml через веб-интерфейс (Add file → Create new file).
# Дальше всё происходит само: пуш → сборка → релиз → приложение само обновляется.

name: Build APK and Release

on:
  push:
    branches: [main]
  workflow_dispatch:

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 20
      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: 17
      - run: npm install
      # Постоянный ключ подписи. Без него каждый билд подписывался бы НОВЫМ ключом,
      # и Android отказывался бы ставить обновление поверх (конфликт подписей).
      # Секрет KEYSTORE_BASE64 — это base64 от debug.keystore (см. шаги в README.md).
      # Если секрета нет или в нём не тот ключ — падаем сразу с понятной ошибкой,
      # чтобы зря не жечь минуты CI и не плодить битые релизы.
      - run: |
          mkdir -p ~/.android
          if [ -z "${{ secrets.KEYSTORE_BASE64 }}" ]; then
            echo "::error::Секрет KEYSTORE_BASE64 пуст или не виден! Создай его: Settings → Secrets and variables → Actions → New repository secret. Значение — содержимое KEYSTORE_BASE64.txt."
            exit 1
          fi
          echo "${{ secrets.KEYSTORE_BASE64 }}" | base64 -d > ~/.android/debug.keystore
          KSHA=$(sha256sum ~/.android/debug.keystore | cut -d' ' -f1)
          echo "Размер файла ключа: $(stat -c%s ~/.android/debug.keystore) байт"
          echo "SHA256 файла ключа: $KSHA"
          echo "Ожидалось:          0d9fcf5a1c3d5035e0b07d76b8e53cc0b044625805ca1ecdbae92145e6e29a87"
          INFO=$(keytool -list -keystore ~/.android/debug.keystore -storepass android -alias androiddebugkey -v 2>&1 || true)
          echo "$INFO" | grep -E "Owner|SHA256" || true
          FP=$(echo "$INFO" | grep -oE 'SHA-?256: [0-9A-F:]+' | head -1 || true)
          case "$FP" in
            *"0C:69:01:F4:67:E0:FD:69"*) echo "Подпись: постоянный ключ FinAnalyzer (отпечаток совпал)" ;;
            *) echo "::error::В секрете НЕ ТОТ ключ! В секрете лежит: ${FP:-нечитаемый файл}, sha256 файла: $KSHA. Нужен ключ из KEYSTORE_BASE64.txt (отпечаток 0C:69:01:F4…)."
               exit 1 ;;
          esac
      # Версия релиза: v<версия-из-package.json>-b<номер-сборки>
      - id: ver
        run: echo "tag=v$(node -p "require('./package.json').version")-b${{ github.run_number }}" >> "$GITHUB_OUTPUT"
      # Вшиваем полную версию со сборкой в приложение, иначе автообновление будет
      # бесконечно предлагать "новую" версию (приложение не знало свой build-номер)
      - run: sed -i "s|^export const APP_VERSION = '.*';|export const APP_VERSION = '${{ steps.ver.outputs.tag }}';|" www/js/config.js
      - run: grep "APP_VERSION" www/js/config.js
      - run: npx cap add android      # создаёт android/ (в .gitignore)
      # ЯВНО прописываем наш ключ подписи в app/build.gradle.
      # Почему так: gradle на раннере упорно игнорирует ~/.android/debug.keystore
      # и подписывает свежесгенерированным ключом — поэтому не полагаемся на
      # неявный поиск ключа, а указываем файл прямо в проекте.
      - run: |
          cp ~/.android/debug.keystore android/app/debug.keystore
          cmp ~/.android/debug.keystore android/app/debug.keystore && echo "Ключ скопирован в проект, байты совпадают"
          python3 - <<'EOF'
          p = 'android/app/build.gradle'
          s = open(p, encoding='utf-8').read()
          assert 'signingConfigs' not in s, 'в шаблоне уже есть signingConfigs — надо посмотреть глазами'
          block = """android {
              signingConfigs {
                  debug {
                      storeFile file('debug.keystore')
                      storePassword 'android'
                      keyAlias 'androiddebugkey'
                      keyPassword 'android'
                  }
              }"""
          assert 'android {' in s
          s = s.replace('android {', block, 1)
          assert 'buildTypes {' in s
          s = s.replace('buildTypes {',
                        'buildTypes {\n        debug {\n            signingConfig signingConfigs.debug\n        }', 1)
          open(p, 'w', encoding='utf-8').write(s)
          print('signingConfigs.debug внедрён и привязан к buildTypes.debug')
          EOF
          grep -A8 "signingConfigs" android/app/build.gradle
      # Подкладываем нативный модуль чтения уведомлений и свою MainActivity
      - run: |
          mkdir -p android/app/src/main/java/com/finanalyzer/app
          cp native/NotificationListenerPlugin.java native/BankNotificationService.java native/MainActivity.java android/app/src/main/java/com/finanalyzer/app/
      # Регистрируем службу чтения уведомлений в манифесте
      - run: |
          python3 - <<'EOF'
          p = 'android/app/src/main/AndroidManifest.xml'
          s = open(p, encoding='utf-8').read()
          svc = ('<service android:name=".BankNotificationService" '
                 'android:label="FinAnalyzer" '
                 'android:permission="android.permission.BIND_NOTIFICATION_LISTENER_SERVICE" '
                 'android:exported="true">'
                 '<intent-filter><action android:name="android.service.notification.NotificationListenerService" /></intent-filter>'
                 '</service>')
          assert '</application>' in s
          s = s.replace('</application>', svc + '</application>')
          open(p, 'w', encoding='utf-8').write(s)
          EOF
      - run: npx cap sync android      # копирует www/ в android-проект
      - run: cd android && chmod +x gradlew && ./gradlew assembleDebug --stacktrace
      # Контроль: APK должен быть подписан нашим постоянным ключом (O=FinAnalyzer).
      # Сначала печатаем реального подписанта — чтобы в логе было видно, чей ключ.
      - run: |
          echo "=== Кем подписан собранный APK ==="
          CERT=$(keytool -printcert -jarfile android/app/build/outputs/apk/debug/app-debug.apk 2>&1 || true)
          echo "$CERT" | grep -E "Owner|SHA256|SHA-256" || echo "$CERT"
          if echo "$CERT" | grep -q "O=FinAnalyzer"; then
            echo "Подпись APK проверена: постоянный ключ"
          else
            SIGNER=$(echo "$CERT" | grep -m1 "Owner" || echo "keytool не смог прочитать подпись")
            echo "::error::APK подписан НЕ постоянным ключом! Подписант: $SIGNER"
            exit 1
          fi
      - uses: actions/upload-artifact@v4
        with:
          name: app-debug
          path: android/app/build/outputs/apk/debug/app-debug.apk
      # Релиз с APK — приложение сверяется с ним и предлагает обновление
      - uses: softprops/action-gh-release@v2
        with:
          tag_name: ${{ steps.ver.outputs.tag }}
          name: FinAnalyzer ${{ steps.ver.outputs.tag }}
          files: android/app/build/outputs/apk/debug/app-debug.apk
